\begin{tikzpicture}[

% Cliente y componentes de apoyo
\node[client] (web) at (0,0) {Aplicación web Angular/Ionic (PWA)\\flujo del votante y consola de gestión};

\node[zone] at (4.0,-6.95) {pg-ballot (PostgreSQL)};
